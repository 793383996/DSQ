## 三、潜在问题排查

### 3.1 性能问题排查

#### ✅ 已解决：死代码问题

- **位置**：`BootstrapService.ts`
- **问题**：约300行未使用的初始化编排代码
- **修复**：已删除
- **影响**：减少打包体积，提高加载速度

#### ✅ 已解决：watch 触发链优化

- **位置**：`demand.ts`, `settings.ts`
- **问题**：初始化时 watch 触发不必要的同步和持久化
- **修复**：添加 `isInitializing` 标记
- **影响**：减少初始化时的不必要操作

#### 🟡 需关注：localStorage 同步写入

- **位置**：`demand.ts:206,214`, `settings.ts:149`
- **问题**：localStorage.setItem() 是同步阻塞操作
- **影响**：数据量大时可能阻塞主线程
- **建议**：

  ```typescript
  // 当前实现
  localStorage.setItem(KEY, JSON.stringify(data))

  // 建议改进：使用 requestIdleCallback 或防抖
  const persistAsync = (key: string, data: unknown) => {
    requestIdleCallback(() => {
      localStorage.setItem(key, JSON.stringify(data))
    })
  }
  ```

#### 🟡 需关注：JSON.stringify 性能

- **位置**：多处 localStorage 操作
- **问题**：频繁的 JSON.stringify 可能影响性能
- **影响**：大数据量时序列化耗时增加
- **建议**：考虑使用更高效的序列化方案或增量更新

#### 🟡 需关注：watch deep 选项

- **位置**：`demand.ts:192`, `settings.ts:174`
- **问题**：`deep: true` 会递归遍历所有属性
- **影响**：大型对象时性能下降
- **建议**：考虑使用 shallowRef 或精确指定监听属性

### 3.2 流程问题排查

#### ✅ 已解决：状态同步方向不明确

- **位置**：`demand.ts`, `App.vue`
- **问题**：初始化时双向同步导致循环

## 五、关键代码实现

### 5.1 DemandStore 状态同步机制

```typescript
// stores/demand.ts

// 关键状态变量
let syncEnabled = false
let lastSyncVersion = 0
let isSyncing = false
let isInitializing = false
let syncDebounceTimer: ReturnType<typeof setTimeout> | null = null
let pendingSyncResolve: (() => void) | null = null
const SYNC_DEBOUNCE_MS = 16

// 单向加载（初始化时使用）
function loadFromLegacy(legacyDemands: DemandItem[], legacyExcludes: string[]): void {
  isInitializing = true
  try {
    demandList.value = legacyDemands.map(d => ({ ...d }))
    excludeList.value = [...legacyExcludes]
    demandVersion.value++
    lastSyncVersion = demandVersion.value
  } finally {
    isInitializing = false
  }
}

// 强制同步（计算前使用）
async function forceSyncToLegacy(): Promise<boolean> {
  if (isInitializing) return false

  if (syncDebounceTimer) {
    clearTimeout(syncDebounceTimer)
    syncDebounceTimer = null
  }

  // 等待正在进行的同步完成
  if (isSyncing) {
    await new Promise<void>(resolve => {
      pendingSyncResolve = resolve
    })
  }

  return doSyncToLegacy(true)
}

// watch 监听（运行时使用）
watch(
  [demandList, excludeList],
  () => {
    if (!isInitializing) {
```

### 5.2 CalculationStore 状态机

```typescript
// stores/calculation.ts

export enum CalcState {
  IDLE = 'idle',
  PREPARING = 'preparing',
  CALCULATING = 'calculating',
  SUCCESS = 'success',
  ERROR = 'error'
}

// 状态转换规则
const transitions: Record<CalcState, CalcState[]> = {
  [CalcState.IDLE]: [CalcState.PREPARING],
  [CalcState.PREPARING]: [CalcState.CALCULATING, CalcState.ERROR],
  [CalcState.CALCULATING]: [CalcState.SUCCESS, CalcState.ERROR],
  [CalcState.SUCCESS]: [CalcState.IDLE, CalcState.PREPARING],
  [CalcState.ERROR]: [CalcState.IDLE, CalcState.PREPARING]
}

function canTransition(to: CalcState): boolean {
  return transitions[state.value].includes(to)
}

function transition(newState: CalcState): boolean {
  if (!canTransition(newState)) {
    logger.warn(`[CalculationStore] Invalid transition: ${state.value} -> ${newState}`)
    return false
  }
  state.value = newState
  return true
}

function forceReset(): void {
  state.value = CalcState.IDLE
  resultItems.value = []
  error.value = null
}
```

### 5.3 InitializationService 状态机

```typescript
// core/services/InitializationService.ts

export enum InitState {
  IDLE = 'IDLE',
  LOADING_RECIPES = 'LOADING_RECIPES',
  LOADING_ICONS = 'LOADING_ICONS',
  BUILDING_INDEX = 'BUILDING_INDEX',
  SYNCING_LEGACY = 'SYNCING_LEGACY',
  READY = 'READY',
  ERROR = 'ERROR'
}

// 计算就绪检查
isCalculationReady(): { ready: boolean; reason?: string } {
  if (this.currentState !== InitState.READY) {
    return { ready: false, reason: `Initialization not complete: ${this.currentState}` }
  }

  const win = window as any

  if (typeof win.find !== 'function') {
    return { ready: false, reason: 'Recipe find function not available' }
  }

  if (!Array.isArray(win.data) || win.data.length === 0) {
    return { ready: false, reason: 'Recipe data not loaded' }
  }

  if (!win.recipeIndexByProduct || Object.keys(win.recipeIndexByProduct).length === 0) {
    return { ready: false, reason: 'Recipe index not built' }
  }

  return { ready: true }
}
```
